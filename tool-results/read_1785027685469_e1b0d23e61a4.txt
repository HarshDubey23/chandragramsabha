   169→// Recharts custom tooltip for enrollment — declared OUTSIDE the component function
   170→// to satisfy react-hooks/static-components lint rule
   171→function EnrollmentTooltip({
   172→  active,
   173→  payload,
   174→  locale,
   175→}: {
   176→  active?: boolean
   177→  payload?: Array<{ value: number; payload: { class: string; classEn: string; boys: number; girls: number; total: number } }>
   178→  locale: 'hi' | 'en'
   179→}) {
   180→  if (active && payload && payload.length) {
   181→    const d = payload[0].payload
   182→    const cls = locale === 'hi' ? d.class : d.classEn
   183→    const boysLbl = locale === 'hi' ? 'लड़के' : 'Boys'
   184→    const girlsLbl = locale === 'hi' ? 'लड़कियां' : 'Girls'
   185→    const totalLbl = locale === 'hi' ? 'कुल' : 'Total'
   186→    return (
   187→      <div className="bg-card border border-border rounded-md px-3 py-2 text-xs shadow-md">
   188→        <div className="font-medium">{cls}</div>
   189→        <div className="text-muted-foreground">{boysLbl}: {d.boys} · {girlsLbl}: {d.girls}</div>
   190→        <div className="font-semibold">{totalLbl}: {d.total}</div>
   191→      </div>
   192→    )
   193→  }
   194→  return null
   195→}
