   169→   169→// Recharts custom tooltip for enrollment — declared OUTSIDE the component function
   170→   170→// to satisfy react-hooks/static-components lint rule
   171→   171→function EnrollmentTooltip({
   172→   172→  active,
   173→   173→  payload,
   174→   174→  locale,
   175→   175→}: {
   176→   176→  active?: boolean
   177→   177→  payload?: Array<{ value: number; payload: { class: string; classEn: string; boys: number; girls: number; total: number } }>
   178→   178→  locale: 'hi' | 'en'
   179→   179→}) {
   180→   180→  if (active && payload && payload.length) {
   181→   181→    const d = payload[0].payload
   182→   182→    const cls = locale === 'hi' ? d.class : d.classEn
   183→   183→    const boysLbl = locale === 'hi' ? 'लड़के' : 'Boys'
   184→   184→    const girlsLbl = locale === 'hi' ? 'लड़कियां' : 'Girls'
   185→   185→    const totalLbl = locale === 'hi' ? 'कुल' : 'Total'
   186→   186→    return (
   187→   187→      <div className="bg-card border border-border rounded-md px-3 py-2 text-xs shadow-md">
   188→   188→        <div className="font-medium">{cls}</div>
   189→   189→        <div className="text-muted-foreground">{boysLbl}: {d.boys} · {girlsLbl}: {d.girls}</div>
   190→   190→        <div className="font-semibold">{totalLbl}: {d.total}</div>
   191→   191→      </div>
   192→   192→    )
   193→   193→  }
   194→   194→  return null
   195→   195→}
